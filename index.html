SELECT aptCode
FROM (
  SELECT aptCode,
    (6371 * ACOS(
       COS(RADIANS(#{lat})) * COS(RADIANS(lat)) * COS(RADIANS(lng) - RADIANS(#{lng}))
     + SIN(RADIANS(#{lat})) * SIN(RADIANS(lat))
    )) AS distance  -- km
  FROM houseinfo
) AS subquery
WHERE distance < #{dis}
